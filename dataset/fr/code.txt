Structures et implémentation :

struct Point {
    x: f64,
    y: f64,
}

impl Point {
    fn new(x: f64, y: f64) -> Self {
        Point { x, y }
    }

    fn distance(&self, other: &Point) -> f64 {
        let dx = self.x - other.x;
        let dy = self.y - other.y;
        (dx * dx + dy * dy).sqrt()
    }
}

fn main() {
    let a = Point::new(0.0, 0.0);
    let b = Point::new(3.0, 4.0);
    println!("distance = {}", a.distance(&b));
}

Enum et pattern matching :

enum Forme {
    Cercle(f64),
    Rectangle(f64, f64),
    Triangle(f64, f64, f64),
}

fn aire(f: &Forme) -> f64 {
    match f {
        Forme::Cercle(r) => std::f64::consts::PI * r * r,
        Forme::Rectangle(l, h) => l * h,
        Forme::Triangle(a, b, c) => {
            let s = (a + b + c) / 2.0;
            (s * (s - a) * (s - b) * (s - c)).sqrt()
        }
    }
}

fn main() {
    let formes = vec![
        Forme::Cercle(5.0),
        Forme::Rectangle(4.0, 6.0),
        Forme::Triangle(3.0, 4.0, 5.0),
    ];
    for f in &formes {
        println!("aire = {:.2}", aire(f));
    }
}

Traits et generics :

trait Surface {
    fn aire(&self) -> f64;
    fn perimetre(&self) -> f64;
}

struct Disque { rayon: f64 }
struct Carre { cote: f64 }

impl Surface for Disque {
    fn aire(&self) -> f64 { std::f64::consts::PI * self.rayon * self.rayon }
    fn perimetre(&self) -> f64 { 2.0 * std::f64::consts::PI * self.rayon }
}

impl Surface for Carre {
    fn aire(&self) -> f64 { self.cote * self.cote }
    fn perimetre(&self) -> f64 { 4.0 * self.cote }
}

fn afficher(s: &dyn Surface) {
    println!("aire={:.2} perimetre={:.2}", s.aire(), s.perimetre());
}

fn main() {
    let d = Disque { rayon: 3.0 };
    let c = Carre { cote: 4.0 };
    afficher(&d);
    afficher(&c);
}

use std::sync::mpsc;

fn main() {
    let (tx, rx) = mpsc::channel();

    for i in 0..5 {
        let tx = tx.clone();
        thread::spawn(move || {
            tx.send(i * i).unwrap();
        });
    }

    let mut resultats: Vec<i32> = rx.iter().collect();
    resultats.sort();
    println!("{resultats:?}");

use std::sync::{Arc, Mutex};
use std::thread;

fn main() {
    let compteur = Arc::new(Mutex::new(0));
    let mut handles = Vec::new();

    for _ in 0..8 {
        let c = Arc::clone(&compteur);
        handles.push(thread::spawn(move || {
            *c.lock().unwrap() += 1;

Pattern builder :

struct RequeteBuilder {
    url: String,
    timeout_ms: u64,
    retries: u32,
}

impl RequeteBuilder {
    fn new(url: &str) -> Self {
        RequeteBuilder {
            url: url.to_string(),
            timeout_ms: 5000,
            retries: 3,
        }
    }

    fn timeout(mut self, ms: u64) -> Self {
        self.timeout_ms = ms;
        self
    }

    fn retries(mut self, n: u32) -> Self {
        self.retries = n;
        self
    }

    fn build(self) -> String {
        format!("GET {} (timeout={}ms, retries={})", self.url, self.timeout_ms, self.retries)
    }
}

fn main() {
    let r = RequeteBuilder::new("https://api.exemple.fr")
        .timeout(10_000)
        .retries(5)
        .build();
    println!("{r}");
}

Erreur personnalisée :

use std::fmt;

#[derive(Debug)]
enum ErreurApp {
    Io(std::io::Error),
    Parse(std::num::ParseIntError),
    Logique(String),
}

impl fmt::Display for ErreurApp {
    fn fmt(&self, f: &mut fmt::Formatter) -> fmt::Result {
        match self {
            Self::Io(e) => write!(f, "IO : {e}"),
            Self::Parse(e) => write!(f, "parse : {e}"),
            Self::Logique(s) => write!(f, "logique : {s}"),
        }
    }
}

impl std::error::Error for ErreurApp {}

impl From<std::io::Error> for ErreurApp {
    fn from(e: std::io::Error) -> Self { Self::Io(e) }
}

impl From<std::num::ParseIntError> for ErreurApp {
    fn from(e: std::num::ParseIntError) -> Self { Self::Parse(e) }
}

fn charger(path: &str) -> Result<i32, ErreurApp> {
    let s = std::fs::read_to_string(path)?;
    let n: i32 = s.trim().parse()?;
    if n < 0 { return Err(ErreurApp::Logique("valeur négative".into())); }
    Ok(n)
}

Lifetime dans une struct :

struct Analyseur<'a> {
    source: &'a str,
    position: usize,
}

impl<'a> Analyseur<'a> {
    fn new(source: &'a str) -> Self {
        Analyseur { source, position: 0 }
    }

    fn token_suivant(&mut self) -> Option<&'a str> {
        let debut = self.source[self.position..]
            .find(|c: char| !c.is_whitespace())
            .map(|i| self.position + i)?;
        let fin = self.source[debut..]
            .find(|c: char| c.is_whitespace())
            .map(|i| debut + i)
            .unwrap_or(self.source.len());
        self.position = fin;
        Some(&self.source[debut..fin])
    }
}

fn main() {
    let mut a = Analyseur::new("bonjour monde rust");
    while let Some(tok) = a.token_suivant() {
        println!("{tok}");
    }
}

Comptage de mots avec HashMap :

use std::collections::HashMap;

fn compter_mots(texte: &str) -> HashMap<&str, usize> {
    let mut compteur = HashMap::new();
    for mot in texte.split_whitespace() {
        *compteur.entry(mot).or_insert(0) += 1;
    }
    compteur
}

fn main() {
    let texte = "le rust est rapide le rust est sûr rust rust";
    let mut freq: Vec<_> = compter_mots(texte).into_iter().collect();
    freq.sort_by(|a, b| b.1.cmp(&a.1));
    for (mot, n) in &freq {
        println!("{mot}: {n}");
    }
}

use std::fs::File;
use std::io::{BufRead, BufReader};

fn main() -> std::io::Result<()> {
    let fichier = File::open("donnees.txt")?;
    let lecteur = BufReader::new(fichier);
    let mut somme = 0i64;
    let mut lignes = 0;

    for ligne in lecteur.lines() {
        let l = ligne?;
        if let Ok(n) = l.trim().parse::<i64>() {

            lignes += 1;
        }
    }

}

pub fn factorielle(n: u64) -> u64 {
    (1..=n).product()
}

pub fn fibonacci(n: u32) -> u64 {
    match n {
        0 => 0,
        1 => 1,
        _ => {
            let (mut a, mut b) = (0u64, 1u64);
            for _ in 2..=n { let c = a + b; a = b; b = c; }
            b
        }
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn test_fact() {
        assert_eq!(factorielle(0), 1);
        assert_eq!(factorielle(5), 120);
        assert_eq!(factorielle(10), 3_628_800);
    }

    #[test]
    fn test_fib() {
        assert_eq!(fibonacci(0), 0);
        assert_eq!(fibonacci(1), 1);
        assert_eq!(fibonacci(10), 55);
    }
}

Opérateurs surchargés :

use std::ops::{Add, Mul, Neg};

#[derive(Debug, Clone, Copy, PartialEq)]
struct Vec2 { x: f64, y: f64 }

impl Vec2 {
    fn new(x: f64, y: f64) -> Self { Vec2 { x, y } }
    fn norme(&self) -> f64 { (self.x * self.x + self.y * self.y).sqrt() }
    fn dot(&self, o: Vec2) -> f64 { self.x * o.x + self.y * o.y }
}

impl Add for Vec2 {
    type Output = Vec2;
    fn add(self, o: Vec2) -> Vec2 { Vec2::new(self.x + o.x, self.y + o.y) }
}

impl Neg for Vec2 {
    type Output = Vec2;
    fn neg(self) -> Vec2 { Vec2::new(-self.x, -self.y) }
}

impl Mul<f64> for Vec2 {
    type Output = Vec2;
    fn mul(self, s: f64) -> Vec2 { Vec2::new(self.x * s, self.y * s) }
}

fn main() {
    let a = Vec2::new(1.0, 2.0);
    let b = Vec2::new(3.0, 4.0);
    println!("{:?}", a + b);
    println!("norme b = {:.4}", b.norme());
    println!("dot = {}", a.dot(b));
}

Atomics sans mutex :

use std::sync::atomic::{AtomicU64, Ordering};
use std::sync::Arc;
use std::thread;

fn main() {
    let compteur = Arc::new(AtomicU64::new(0));
    let mut handles = Vec::new();

    for _ in 0..8 {
        let c = Arc::clone(&compteur);
        handles.push(thread::spawn(move || {
            for _ in 0..1000 {
                c.fetch_add(1, Ordering::Relaxed);
            }
        }));
    }

    for h in handles { h.join().unwrap(); }
    println!("total = {}", compteur.load(Ordering::Relaxed));
}

#[derive(Debug)]
struct Etudiant {
    nom: String,
    note: f64,
}

fn main() {
    let mut etudiants = vec![
        Etudiant { nom: "Alice".into(), note: 17.5 },
        Etudiant { nom: "Bob".into(), note: 14.0 },
        Etudiant { nom: "Charlie".into(), note: 19.0 },
        Etudiant { nom: "Diana".into(), note: 14.0 },

        b.note.partial_cmp(&a.note).unwrap()
            .then(a.nom.cmp(&b.nom))
    });

        println!("{}: {}", e.nom, e.note);
    }
}

Cow pour éviter les allocations :

use std::borrow::Cow;

fn nettoyer(s: &str) -> Cow<str> {
    if s.chars().all(|c| c.is_ascii() && !c.is_control()) {
        Cow::Borrowed(s)
    } else {
        Cow::Owned(s.chars().filter(|c| c.is_ascii() && !c.is_control()).collect())
    }
}

fn main() {
    let propre = nettoyer("bonjour monde");
    let sale = nettoyer("bonjour\x00monde\x1b");
    println!("{propre}");
    println!("{sale}");
}

fn main() {
    let donnees = vec![1i64, 2, 3, 4, 5, 6, 7, 8];
    let milieu = donnees.len() / 2;
    let (gauche, droite) = donnees.split_at(milieu);

    std::thread::scope(|s| {
        let h1 = s.spawn(|| gauche.iter().sum::<i64>());
        let h2 = s.spawn(|| droite.iter().sum::<i64>());
        println!("somme = {}", h1.join().unwrap() + h2.join().unwrap());
    });
}
