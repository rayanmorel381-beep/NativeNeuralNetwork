Tutoriels pratiques Rust

Tutoriel 1 : Premier projet Cargo. Installe Rust via rustup curl --proto '=https' --tlsv1.2 -sSf https://sh.rustup.rs | sh. Vérifie l'installation avec rustc --version et cargo --version. Crée un projet : cargo new bonjour. Le répertoire bonjour contient Cargo.toml et src/main.rs avec un programme par défaut. cargo run compile et exécute. cargo build produit le binaire dans target/debug. cargo build --release produit la version optimisée dans target/release. Modifie src/main.rs pour personnaliser le message et observe la recompilation incrémentale.

Tutoriel 2 : Lire l'entrée utilisateur. use std::io::{self, BufRead, Write}; fn main() { print!("Quel est ton nom ? "); io::stdout().flush().unwrap(); let stdin = io::stdin(); let mut ligne = String::new(); stdin.lock().read_line(&mut ligne).unwrap(); let nom = ligne.trim(); println!("Bonjour, {nom} !"); }. read_line conserve le \n final, trim l'élimine. flush force l'affichage du prompt avant la lecture.

Tutoriel 3 : Manipuler des collections. Crée un Vec de noms. let mut noms = vec![String::from("Alice"), String::from("Bob"), String::from("Charlie")]; noms.push(String::from("David")); noms.sort(); for n in &noms { println!("{n}"); }. Filtre par longueur : let longs: Vec<&String> = noms.iter().filter(|n| n.len() > 4).collect();. Transformation : let majuscules: Vec<String> = noms.iter().map(|n| n.to_uppercase()).collect();.

Tutoriel 4 : Structures et méthodes. struct Cercle { rayon: f64 } impl Cercle { fn nouveau(r: f64) -> Self { Cercle { rayon: r } } fn aire(&self) -> f64 { std::f64::consts::PI * self.rayon.powi(2) } fn perimetre(&self) -> f64 { 2.0 * std::f64::consts::PI * self.rayon } } fn main() { let c = Cercle::nouveau(5.0); println!("aire: {:.2}", c.aire()); println!("périmètre: {:.2}", c.perimetre()); }.

Tutoriel 5 : Énumérations et pattern matching. enum Forme { Cercle(f64), Carre(f64), Rectangle { largeur: f64, hauteur: f64 } } fn aire(f: &Forme) -> f64 { match f { Forme::Cercle(r) => std::f64::consts::PI * r * r, Forme::Carre(c) => c * c, Forme::Rectangle { largeur, hauteur } => largeur * hauteur } } fn main() { let formes = vec![Forme::Cercle(3.0), Forme::Carre(4.0), Forme::Rectangle { largeur: 5.0, hauteur: 6.0 }]; for f in &formes { println!("{:.2}", aire(f)); } }.

Tutoriel 6 : Gestion d'erreurs avec Result. use std::num::ParseIntError; fn doubler(s: &str) -> Result<i32, ParseIntError> { let n: i32 = s.parse()?; Ok(n * 2) } fn main() { match doubler("21") { Ok(n) => println!("résultat: {n}"), Err(e) => println!("erreur: {e}") } match doubler("abc") { Ok(n) => println!("résultat: {n}"), Err(e) => println!("erreur: {e}") } }. Le ? propage automatiquement l'erreur si parse échoue.

Tutoriel 7 : Lire un fichier. use std::fs; fn main() -> std::io::Result<()> { let contenu = fs::read_to_string("notes.txt")?; for ligne in contenu.lines() { if ligne.starts_with("#") { continue; } println!("> {ligne}"); } Ok(()) }. La signature main retournant Result permet d'utiliser ? directement.

Tutoriel 8 : Écrire dans un fichier. use std::io::Write; fn main() -> std::io::Result<()> { let mut f = std::fs::File::create("log.txt")?; for i in 0..10 { writeln!(f, "entrée {i}")?; } Ok(()) }. Pour append, std::fs::OpenOptions::new().append(true).open(path).

Tutoriel 9 : HashMap basique. use std::collections::HashMap; fn main() { let mut scores = HashMap::new(); scores.insert(String::from("Alice"), 95); scores.insert(String::from("Bob"), 87); for (nom, score) in &scores { println!("{nom}: {score}"); } if let Some(s) = scores.get("Alice") { println!("score d'Alice: {s}"); } *scores.entry(String::from("Charlie")).or_insert(0) += 10; }. or_insert crée la clé si absente.

Tutoriel 10 : Trait de base. trait Animal { fn nom(&self) -> &str; fn son(&self) -> &str; fn presentation(&self) { println!("Je suis {} et je fais {}", self.nom(), self.son()); } } struct Chien; impl Animal for Chien { fn nom(&self) -> &str { "Rex" } fn son(&self) -> &str { "Ouaf" } } struct Chat; impl Animal for Chat { fn nom(&self) -> &str { "Minou" } fn son(&self) -> &str { "Miaou" } } fn main() { let animaux: Vec<Box<dyn Animal>> = vec![Box::new(Chien), Box::new(Chat)]; for a in &animaux { a.presentation(); } }.

Tutoriel 11 : Iterators avancés. fn main() { let n: Vec<i32> = (1..=100).collect(); let pairs_carres: Vec<i32> = n.iter().filter(|&&x| x % 2 == 0).map(|&x| x * x).take(10).collect(); println!("{:?}", pairs_carres); let somme: i32 = n.iter().filter(|&&x| x % 3 == 0).sum(); println!("somme multiples de 3: {somme}"); let max = n.iter().max().unwrap(); let min = n.iter().min().unwrap(); println!("max: {max}, min: {min}"); }.

Tutoriel 12 : Closures avec capture. fn main() { let prefixe = String::from("[LOG]"); let logger = |msg: &str| { println!("{prefixe} {msg}"); }; logger("démarrage"); logger("traitement"); logger("fin"); let mut compteur = 0; let mut incrementer = || { compteur += 1; println!("appel {compteur}"); }; incrementer(); incrementer(); incrementer(); }.

Tutoriel 13 : Threads et channels. use std::sync::mpsc; use std::thread; fn main() { let (tx, rx) = mpsc::channel(); for i in 0..5 { let tx = tx.clone(); thread::spawn(move || { tx.send(format!("message de {i}")).unwrap(); }); } drop(tx); for msg in rx { println!("reçu: {msg}"); } }. drop du sender d'origine indispensable pour que rx termine.

Tutoriel 14 : Arc et Mutex pour partage thread. use std::sync::{Arc, Mutex}; use std::thread; fn main() { let compteur = Arc::new(Mutex::new(0)); let mut handles = vec![]; for _ in 0..10 { let c = Arc::clone(&compteur); handles.push(thread::spawn(move || { for _ in 0..1000 { *c.lock().unwrap() += 1; } })); } for h in handles { h.join().unwrap(); } println!("compteur final: {}", *compteur.lock().unwrap()); }.

Tutoriel 15 : Premier projet async avec tokio. Ajoute tokio dans Cargo.toml : tokio = { version = "1", features = ["full"] }. use tokio::time::{sleep, Duration}; #[tokio::main] async fn main() { println!("début"); sleep(Duration::from_secs(1)).await; println!("après 1 seconde"); let h1 = tokio::spawn(tache(1)); let h2 = tokio::spawn(tache(2)); tokio::join!(h1, h2); println!("fin"); } async fn tache(id: u32) { sleep(Duration::from_millis(500)).await; println!("tâche {id} terminée"); }.

Tutoriel 16 : Serveur HTTP minimal avec axum. Cargo.toml : axum = "0.7", tokio = { version = "1", features = ["full"] }. use axum::{routing::get, Router}; #[tokio::main] async fn main() { let app = Router::new().route("/", get(|| async { "Bonjour depuis Axum" })).route("/echo/:msg", get(|axum::extract::Path(msg): axum::extract::Path<String>| async move { format!("Tu as dit: {msg}") })); let listener = tokio::net::TcpListener::bind("127.0.0.1:3000").await.unwrap(); println!("serveur sur http://127.0.0.1:3000"); axum::serve(listener, app).await.unwrap(); }.

Tutoriel 17 : Client HTTP avec ureq. Pour la simplicité sync, ajoute ureq = "2". fn main() -> Result<(), Box<dyn std::error::Error>> { let reponse = ureq::get("https://api.github.com/zen").call()?; let texte = reponse.into_string()?; println!("Zen de GitHub: {texte}"); Ok(()) }. Pour async, reqwest est l'équivalent.

Tutoriel 18 : Parsing d'arguments CLI avec clap. Cargo.toml : clap = { version = "4", features = ["derive"] }. use clap::Parser; #[derive(Parser)] struct Args { #[arg(short, long)] nom: String, #[arg(short, long, default_value_t = 1)] repetitions: u32 } fn main() { let args = Args::parse(); for _ in 0..args.repetitions { println!("Bonjour, {} !", args.nom); } }. Lance avec cargo run -- --nom Alice --repetitions 3.

Tutoriel 19 : Sérialisation JSON sans dépendance externe. Implémente un mini parser pour des objets plats. fn parse_string(input: &str) -> Option<String> { let s = input.trim().strip_prefix('"')?.strip_suffix('"')?; Some(s.to_string()) }. Pour les vrais besoins, serde + serde_json sont la norme.

Tutoriel 20 : Tests unitaires et organisation. fn additionner(a: i32, b: i32) -> i32 { a + b } #[cfg(test)] mod tests { use super::*; #[test] fn test_positifs() { assert_eq!(additionner(2, 3), 5); } #[test] fn test_negatifs() { assert_eq!(additionner(-1, -2), -3); } #[test] fn test_zero() { assert_eq!(additionner(0, 0), 0); } #[test] #[should_panic(expected = "overflow")] fn test_overflow() { additionner(i32::MAX, 1); } }. cargo test exécute toutes les fonctions test.

Tutoriel 21 : Benchmark avec criterion. Cargo.toml : [dev-dependencies] criterion = "0.5"; [[bench]] name = "mon_bench" harness = false. benches/mon_bench.rs : use criterion::{criterion_group, criterion_main, Criterion}; fn fib(n: u64) -> u64 { if n < 2 { n } else { fib(n - 1) + fib(n - 2) } } fn benchmark(c: &mut Criterion) { c.bench_function("fib 20", |b| b.iter(|| fib(20))); } criterion_group!(benches, benchmark); criterion_main!(benches);. cargo bench produit un rapport HTML détaillé.

Tutoriel 22 : Workspace multi-crate. Cargo.toml racine : [workspace] members = ["app", "core", "utils"]. Crée chaque membre : cargo new --lib core, cargo new --lib utils, cargo new app. Le binaire app dépend de core et utils via path = "../core" dans son Cargo.toml. Une seule compilation pour tout le workspace, target/ et Cargo.lock partagés.

Tutoriel 23 : Documentation et publication. Annote tes fonctions publiques avec ///. Inclus un exemple : /// Calcule la somme des deux entiers. /// /// # Exemple /// /// ``` /// assert_eq!(mon_crate::somme(2, 3), 5); /// ```. cargo doc --open génère et ouvre la doc HTML. cargo test exécute les doctests. Pour publier : compte sur crates.io, cargo login <token>, cargo publish.

Tutoriel 24 : Profilage et optimisation. Compile en release avec cargo build --release. Pour profiler avec cargo flamegraph (Linux/macOS, nécessite perf ou dtrace) : cargo flamegraph --bin mon_app. Ouvre flamegraph.svg dans le navigateur. Identifie les fonctions hot, vérifie si l'inlining manque, examine les allocations. cargo bench compare avant/après modifications.

Tutoriel 25 : Cross-compilation. Liste les targets : rustup target list. Ajoute une target : rustup target add aarch64-unknown-linux-gnu. Configure le linker dans .cargo/config.toml : [target.aarch64-unknown-linux-gnu] linker = "aarch64-linux-gnu-gcc". Compile : cargo build --target aarch64-unknown-linux-gnu --release. Le binaire est dans target/aarch64-unknown-linux-gnu/release/.

Tutoriel 26 : Trait avancés et blanket implémentations. Un trait peut avoir des méthodes par défaut : trait Affichable { fn afficher(&self) -> String; fn afficher_en_majuscules(&self) -> String { self.afficher().to_uppercase() } }. La deuxième méthode est gratuite pour tous les implémenteurs. Un blanket impl couvre tous les types satisfaisant une contrainte : impl<T: std::fmt::Display> Affichable for T { fn afficher(&self) -> String { format!("{}", self) } }.

Tutoriel 27 : Fermetures et traits Fn. Les closures en Rust implémentent un ou plusieurs des traits Fn, FnMut, FnOnce selon leur capture. let s = String::from("hi"); let c1 = || println!("{s}"); // Fn, capture &s. let mut n = 0; let mut c2 = || { n += 1; }; // FnMut, capture &mut n. let s2 = String::from("move"); let c3 = move || println!("{s2}"); // FnOnce ou Fn selon contenu, possède s2. Les paramètres de fonction s'écrivent fn appliquer<F: Fn(i32) -> i32>(f: F, x: i32) -> i32 { f(x) }.

Tutoriel 28 : Itérateurs personnalisés. Implémente Iterator pour tes propres types. struct Intervalle { courant: i32, fin: i32, pas: i32 } impl Iterator for Intervalle { type Item = i32; fn next(&mut self) -> Option<i32> { if self.courant < self.fin { let val = self.courant; self.courant += self.pas; Some(val) } else { None } } } Utilisation : for n in (Intervalle { courant: 0, fin: 10, pas: 2 }) { println!("{n}"); } Toutes les méthodes adaptateurs (map, filter, take, collect...) sont disponibles gratuitement.

Tutoriel 29 : Durées de vie (lifetimes) explicites. fn plus_long<'a>(x: &'a str, y: &'a str) -> &'a str { if x.len() >= y.len() { x } else { y } } Le paramètre 'a lie les lifetimes des deux entrées et de la sortie. Si x et y ont des lifetimes différentes, le compilateur prend le minimum. Dans les structs : struct Extrait<'a> { contenu: &'a str } — la struct ne peut pas vivre plus longtemps que la &str qu'elle contient.

Tutoriel 30 : Tests d'intégration. Place tes tests d'intégration dans le répertoire tests/ à la racine du crate. Chaque fichier est un crate de test indépendant. Exemple tests/calcul_test.rs : use mon_crate::additionner; #[test] fn integration_base() { assert_eq!(additionner(10, 20), 30); } cargo test exécute aussi ces tests. Pour des helpers partagés entre plusieurs fichiers de tests, crée tests/common/mod.rs. Le module est disponible via use common::prepare;.

Tutoriel 31 : Concurrence avec scoped threads. Depuis Rust 1.63, std::thread::scope permet de spawner des threads qui empruntent des données locales. let data = vec![1, 2, 3, 4, 5, 6]; let chunk_size = data.len() / 2; std::thread::scope(|s| { let (gauche, droite) = data.split_at(chunk_size); s.spawn(|| println!("gauche: {:?}", gauche)); s.spawn(|| println!("droite: {:?}", droite)); }); Le scope garantit que tous les threads se terminent avant la fin du bloc.

Tutoriel 32 : Gestion des processus. use std::process::Command; let output = Command::new("ls").arg("-la").arg("/tmp").output().expect("ls échoué"); println!("stdout: {}", String::from_utf8_lossy(&output.stdout)); println!("stderr: {}", String::from_utf8_lossy(&output.stderr)); Pour du streaming : utiliser spawn() au lieu de output(), puis piping via stdin/stdout handles.

Tutoriel 33 : Sérialisation avec serde (avec dépendance). Si tu peux ajouter une dépendance, serde + serde_json est la norme. Cargo.toml : serde = { version = "1", features = ["derive"] }, serde_json = "1". Code : use serde::{Serialize, Deserialize}; #[derive(Serialize, Deserialize, Debug)] struct Point { x: f64, y: f64 } let p = Point { x: 1.0, y: 2.5 }; let json = serde_json::to_string(&p).unwrap(); let p2: Point = serde_json::from_str(&json).unwrap();

Tutoriel 34 : Modules et visibilité. mod géométrie { pub struct Cercle { pub rayon: f64 } impl Cercle { pub fn aire(&self) -> f64 { std::f64::consts::PI * self.rayon * self.rayon } } } use géométrie::Cercle; fn main() { let c = Cercle { rayon: 3.0 }; println!("{:.3}", c.aire()); } Un champ sans pub est privé au module. pub(crate) rend visible dans tout le crate. pub(super) dans le module parent seulement.

Tutoriel 35 : Documentation Rust. Les commentaires /// génèrent de la documentation HTML. Les blocs de code dans la doc sont des doctests exécutés par cargo test. /// Retourne le double de `n`. /// /// # Exemples /// /// ``` /// assert_eq!(mon_crate::doubler(4), 8); /// ``` /// /// # Panics /// /// Ne panique jamais. pub fn doubler(n: i32) -> i32 { n * 2 } Les sections conventionnelles : Exemples, Panics, Errors, Safety. cargo doc --open génère la doc navigable.
